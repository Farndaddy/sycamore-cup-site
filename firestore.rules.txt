// Paste this into Firebase Console > Firestore Database > Rules, then Publish.
// This intentionally has NO login system (any of the guys can enter scores from
// their phone with just the site URL) but blocks deleting data and caps how much
// text can be written per field so a stray bot can't trash the database.

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /scores/{scoreId} {
      allow read: if true;
      allow create, update: if
        request.resource.data.strokes is number &&
        request.resource.data.strokes > 0 &&
        request.resource.data.strokes < 20 &&
        request.resource.data.hole is number &&
        request.resource.data.hole > 0 &&
        request.resource.data.hole <= 18;
      allow delete: if false;
    }

    match /courses/{courseId} {
      allow read: if true;
      allow create, update: if request.resource.data.pars is list;
      allow delete: if false;
    }
  }
}
